FORM f_show_alv  USING  fp_fcat       TYPE lvc_t_fcat

                        fp_liq_transp TYPE ty_t_liqtrans.

  DATA: lit_transp  TYPE ty_t_liqtrans,
        lit_fcat    TYPE lvc_t_fcat,
        lwa_variant TYPE disvariant.

  FIELD-SYMBOLS <lfs_fcat> TYPE lvc_s_fcat.

* @0001 - Ajustes RITM16870750 - INI
  DATA:
    vl_txtcol TYPE string.
* @0001 - Ajustes RITM16870750 - FIN


  lit_transp[]  = fp_liq_transp[].
  lit_fcat      = fp_fcat.

  lwa_variant-report = sy-repid.

  SORT lit_fcat BY fieldname.

  LOOP AT lit_fcat ASSIGNING <lfs_fcat>.
    CLEAR: vl_txtcol.
    CASE <lfs_fcat>-fieldname.
      WHEN 'BOX'.

        <lfs_fcat>-no_out    =   'X'.
        <lfs_fcat>-outputlen =   1.

      WHEN 'TKNUM' OR 'DOC_PROV' OR 'DOC_PROV_ANUL' OR 'MBLNR' OR
            'EBELN_TER' OR 'SALESORD_ALQ' OR  'FACTURA_ALQ' OR
            'EBELN_IC'  OR 'MBLNR_IC' OR 'SALESORD_IC' OR
            'FACTURA_IC'
            OR 'BELNR_IC'.                                  "++DZ0003

        <lfs_fcat>-hotspot  =   'X'.

* @0001 - Ajustes RITM16870750 - INI
      WHEN 'CONS_COMB_REAL'.
        vl_txtcol = 'Grifo'.

      WHEN 'KMVAC'.
        vl_txtcol = 'Combustible Crédito'.

      WHEN 'KMREAL_ALQ'.
        vl_txtcol = 'Combustible Efectivo'.

* @0001 - Ajustes RITM16870750 - FIN
      WHEN OTHERS.
    ENDCASE.

* @0001 - Ajustes RITM16870750 - INI
    IF vl_txtcol IS INITIAL.
      CONTINUE.
    ELSE.
      <lfs_fcat>-reptext   = vl_txtcol.
      <lfs_fcat>-scrtext_s = vl_txtcol.
      <lfs_fcat>-scrtext_m = vl_txtcol.
      <lfs_fcat>-scrtext_l = vl_txtcol.
    ENDIF.
* @0001 - Ajustes RITM16870750 - FIN

  ENDLOOP.

  "Keeping the Consumption of Combustible Editable only for READ option
  IF rb_pliq IS NOT INITIAL.

    READ TABLE lit_fcat ASSIGNING <lfs_fcat> WITH KEY
    fieldname = 'CONS_COMB_REAL' BINARY SEARCH.
    IF sy-subrc IS INITIAL.

      <lfs_fcat>-edit    =   'X'.
      <lfs_fcat>-outputlen = '20'.

    ENDIF.

    READ TABLE lit_fcat ASSIGNING <lfs_fcat> WITH KEY
    fieldname = 'KMVAC' BINARY SEARCH.
    IF sy-subrc IS INITIAL.

      <lfs_fcat>-edit    =   'X'.
      <lfs_fcat>-outputlen = '20'.

    ENDIF.

    READ TABLE lit_fcat ASSIGNING <lfs_fcat> WITH KEY
    fieldname = 'KMREAL_ALQ' BINARY SEARCH.
    IF sy-subrc IS INITIAL.

      <lfs_fcat>-edit    =   'X'.
      <lfs_fcat>-outputlen = '20'.

    ENDIF.

  ENDIF.
  IF rb_liq IS NOT INITIAL.

    READ TABLE lit_fcat ASSIGNING <lfs_fcat> WITH KEY
    fieldname = 'ICON' BINARY SEARCH.
    IF sy-subrc IS INITIAL.

      <lfs_fcat>-col_pos    = '1'.
      <lfs_fcat>-outputlen  = '4'.

    ENDIF.


  ELSE.

    READ TABLE lit_fcat ASSIGNING <lfs_fcat> WITH KEY
    fieldname = 'ICON' BINARY SEARCH.
    IF sy-subrc IS INITIAL.

      <lfs_fcat>-no_out    =   'X'.

    ENDIF.

  ENDIF.

  it_fcat       = lit_fcat.

  SORT it_fcat BY col_pos.

  CALL METHOD o_grid->set_table_for_first_display
    EXPORTING
      is_variant      = lwa_variant
      is_layout       = fp_lay
    CHANGING
      it_fieldcatalog = it_fcat
      it_outtab       = it_liq_transp[].


ENDFORM.                    " F_SHOW_ALV

*----------------------------------------------------------------------*
FORM f_preliq  USING  fp_rows       TYPE lvc_t_row
               CHANGING fp_liq_transp TYPE ty_t_liqtrans.

  FIELD-SYMBOLS: <lfs_row>        TYPE lvc_s_row,
                 <lfs_liq_trans>  TYPE zmmtt_liq_transp,
                 <lfs_liq_trans2> TYPE zmmtt_liq_transp.

  DATA: vl_texto  TYPE char50,
        vl_texto2 TYPE  char50,
        vl_texto3 TYPE  char50.


  LOOP AT fp_rows ASSIGNING <lfs_row>.

    READ TABLE fp_liq_transp ASSIGNING <lfs_liq_trans>
                              INDEX <lfs_row>-index.

    IF sy-subrc IS INITIAL.

      <lfs_liq_trans>-stat_tdlnr = '1'.

      "LHERAZO RQ26
      IF <lfs_liq_trans>-nro_parte IS INITIAL.
        MESSAGE i002(zmmcm_rq26_1227). "Para preliquidar es obligatorio el número de parte
        LEAVE LIST-PROCESSING.
      ELSE.

        IF <lfs_liq_trans>-flota EQ 'P' AND  <lfs_liq_trans>-tarf_cond1 IS INITIAL
          "Solo aplica la validacion para las sociedades Permitidas
          AND ( <lfs_liq_trans>-bukrs IN ir_bukrs_tar AND ir_bukrs_tar[] IS NOT INITIAL ). "MX007608 22.11.2024
          MESSAGE i003(zmmcm_rq26_1227) . "Tarifa del conductor es cero, no se puede preliquidar
          LEAVE LIST-PROCESSING.
        ELSE.
          LOOP AT fp_liq_transp ASSIGNING <lfs_liq_trans2>.
            IF <lfs_liq_trans2>-nro_parte = <lfs_liq_trans>-nro_parte AND sy-tabix NE <lfs_row>-index.
              IF <lfs_liq_trans2>-conduc1 NE <lfs_liq_trans>-conduc1 OR
                 <lfs_liq_trans2>-conduc2 NE <lfs_liq_trans>-conduc2.

                CONCATENATE  TEXT-d16 <lfs_liq_trans2>-nro_parte INTO vl_texto  SEPARATED BY space.
                CONCATENATE  TEXT-d17 <lfs_liq_trans2>-tknum     INTO vl_texto2 SEPARATED BY space.
                CONCATENATE  vl_texto2 ','                       INTO vl_texto2.
                CONCATENATE  <lfs_liq_trans>-tknum TEXT-d18      INTO vl_texto3 SEPARATED BY space.

                MESSAGE  i004(zmmcm_rq26_1227) WITH vl_texto vl_texto2 vl_texto3.
                LEAVE LIST-PROCESSING.

              ENDIF.
            ENDIF.
          ENDLOOP.
        ENDIF.
      ENDIF.



      "FIN LHERAZO RQ26

    ENDIF.

  ENDLOOP.

ENDFORM.                    " F_PRELIQ

           tknum TYPE tknum,

           vbeln TYPE vbeln,
         END     OF  lty_vttp,

         BEGIN   OF  lty_lips,
           vbeln TYPE vbeln_vl,
           posnr TYPE posnr_vl,
           werks TYPE werks_d,
         END     OF  lty_lips,

         BEGIN         OF  lty_cesp1,
           bukrs      TYPE bukrs,

        lit_lips         TYPE STANDARD TABLE OF lty_lips,

        lit_cesp1        TYPE STANDARD TABLE OF lty_cesp1.

                 <lfs_liqt>  TYPE zmmtt_liq_transp,

                 <lfs_vttp>  TYPE lty_vttp,
                 <lfs_lips>  TYPE lty_lips.

  lit_const[] = fp_const[].

              FROM vttp

              FOR ALL ENTRIES IN lit_liqt
              WHERE tknum = lit_liqt-tknum.

            SELECT  bukrs
                    centro
                    shtyp
                    tipo
                    werks_esp
                    werks_base
                    werks_reg
                    lgort
                    kostl_ter
                    kostl_ic
                    FROM zmmtt_tran_cesp1
                    INTO TABLE lit_cesp1
                    FOR ALL ENTRIES IN fp_transp_cbase
                    WHERE bukrs   = lv_bukrs
                    AND   centro  = fp_transp_cbase-cbase
                    AND   shtyp   = fp_transp_cbase-shtyp
                    AND   tipo    = 'T1'.

                            fp_tplst        TYPE tplst
                  CHANGING  fp_werks        TYPE werks_d.

  FIELD-SYMBOLS <lfs_cbase> TYPE  ty_transp_cbase.

    READ TABLE fp_transp_cbase ASSIGNING <lfs_cbase>
                                WITH KEY  tknum = fp_tknum
                                          shtyp = fp_shtyp
                                          tplst = fp_tplst
                                          BINARY SEARCH.

    IF sy-subrc IS INITIAL.

      fp_werks = <lfs_cbase>-cbase.

    ENDIF.

  ENDIF.

ENDFORM.                    " F_GET_CBASE

        lwa_liq TYPE zmmtt_liq_transp.                      "++DZ0005

        <lfs_liq_trans1>-belnr_ic  IS NOT INITIAL.

      lwa_tdlnr_werks-ebeln = <lfs_liq_trans1>-ebeln_ic.

    ENDIF.

  SORT lit_tdlnr_werks BY vbeln.

  IF lit_tdlnr_werks IS NOT INITIAL.
    SELECT  vbeln
            netwr
            xblnr                                           "++DZ0008
            mwsbk
            FROM vbrk
            INTO TABLE lit_vbrk
            FOR ALL ENTRIES IN lit_tdlnr_werks
            WHERE vbeln = lit_tdlnr_werks-vbeln.
    IF sy-subrc IS INITIAL.
      SORT lit_vbrk BY vbeln.
    ENDIF.
  ENDIF.

  lit_const[] = it_ztconst[].
  SORT lit_const BY fieldname grpname.

  READ TABLE lit_const ASSIGNING <lfs_const>
                      WITH KEY  fieldname = 'TEXTOS'
                                grpname   = 'VERIF_IC'

    lv_bktxt = <lfs_const>-line_low.
    lv_sgtxt = <lfs_const>-line_high.
  ENDIF.

  LOOP AT lit_tdlnr_werks ASSIGNING <lfs_tdlnr_werks>.

    lwa_tdlnr_werks = <lfs_tdlnr_werks>.

          " lwa_tdlnr_werks-vbeln. --DZ0008

        EXPORTING

          im_wa_miro_data = lwa_miro
        IMPORTING

          EXPORTING
            wait = 'X'.

        CLEAR lwa_ret.
        lwa_ret-type        = 'S'.
        lwa_ret-id          = 'ZMM'.
        lwa_ret-number      = '174'.
        lwa_ret-message_v1  = lv_belnr.
        APPEND lwa_ret TO lit_ret.

        LOOP AT lit_trans_ic ASSIGNING <lfs_trans_ic>.

          READ TABLE fp_liq_transp ASSIGNING <lfs_liq_trans>
                              WITH KEY  bukrs = <lfs_trans_ic>-bukrs
                                        tknum = <lfs_trans_ic>-tknum
                                        shtyp = <lfs_trans_ic>-shtyp
                                        tplst = <lfs_trans_ic>-tplst

            <lfs_liq_trans>-gjahr_ic  = lv_gjahr.
            <lfs_liq_trans>-nombre_ic = sy-uname.
            <lfs_liq_trans>-fecha_ic  = sy-datum.
            <lfs_liq_trans>-hora_ic   = sy-uzeit.
            <lfs_liq_trans>-stat_ic   = '2'.
            <lfs_liq_trans>-icon      = '@08@'.

            lwa_liq = <lfs_liq_trans>.                      "++DZ0005
            APPEND lwa_liq TO lit_liq.                      "++DZ0005

          ENDIF.

        ENDLOOP.

      ELSE.

        LOOP AT lit_trans_ic ASSIGNING <lfs_trans_ic>.

          READ TABLE fp_liq_transp ASSIGNING <lfs_liq_trans>
                              WITH KEY  bukrs = <lfs_trans_ic>-bukrs
                                        tknum = <lfs_trans_ic>-tknum
                                        shtyp = <lfs_trans_ic>-shtyp
                                        tplst = <lfs_trans_ic>-tplst
                                          BINARY SEARCH.

          IF sy-subrc IS INITIAL.

            <lfs_liq_trans>-icon        = '@0A@'.

            lwa_liq = <lfs_liq_trans>.                      "++DZ0005
            APPEND lwa_liq TO lit_liq.                      "++DZ0005

          ENDIF.

        ENDLOOP.

        CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.

      ENDIF.

      APPEND LINES OF lit_ret TO fp_ret.

      CLEAR:  lv_docfi,lv_docbuk,lv_gjahr,lv_belnr,lwa_miro,lwa_ret.
      REFRESH:  lit_ret,lit_trans_ic.
